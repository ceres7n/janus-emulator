# JanusLauncher.exe - the player-side launcher with a window (see README.md). Built by the root
# CMakeLists.txt (it lands in the project root like the servers), or on its own:
#   cmake -S tools/janus-launcher -B build-launcher -A Win32
#   cmake --build build-launcher --config Release
cmake_minimum_required(VERSION 3.20)

if(CMAKE_CURRENT_SOURCE_DIR STREQUAL CMAKE_SOURCE_DIR)
    project(JanusLauncher VERSION 0.1.0 LANGUAGES CXX)
    set(CMAKE_CXX_STANDARD 17)
    set(CMAKE_CXX_STANDARD_REQUIRED ON)
    # The client is 32-bit; the launcher reads its memory with the 32-bit APIs.
    if(CMAKE_SIZEOF_VOID_P EQUAL 8)
        message(FATAL_ERROR "JanusLauncher must build 32-bit. Reconfigure with: cmake -A Win32")
    endif()
endif()

# FILEVERSION / ProductVersion of the .rc, from the CMake project version.
file(CONFIGURE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/janus_launcher_version.h" CONTENT
"// Generated by tools/janus-launcher/CMakeLists.txt - do not edit.
#ifndef JANUS_LAUNCHER_VERSION_H
#define JANUS_LAUNCHER_VERSION_H
#define JL_VERSION_MAJOR ${PROJECT_VERSION_MAJOR}
#define JL_VERSION_MINOR ${PROJECT_VERSION_MINOR}
#define JL_VERSION_PATCH ${PROJECT_VERSION_PATCH}
#define JL_VERSION_STR \"${PROJECT_VERSION}\"
#endif
")

# /W4 for this target only: drop the /W3 the root adds to every target (else cl warns D9025).
get_property(_jl_opts DIRECTORY PROPERTY COMPILE_OPTIONS)
list(REMOVE_ITEM _jl_opts /W3)
set_property(DIRECTORY PROPERTY COMPILE_OPTIONS ${_jl_opts})

add_executable(JanusLauncher WIN32 janus_launcher.cpp janus_launcher.rc)
target_include_directories(JanusLauncher PRIVATE "${CMAKE_CURRENT_BINARY_DIR}" "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_definitions(JanusLauncher PRIVATE UNICODE _UNICODE)
target_compile_options(JanusLauncher PRIVATE $<$<COMPILE_LANGUAGE:CXX>:/W4 /EHsc>)
target_link_libraries(JanusLauncher PRIVATE bcrypt comctl32 comdlg32 ole32 shell32 ws2_32)

# Static CRT: only Windows system DLLs, so players install no Visual C++ redistributable.
set_target_properties(JanusLauncher PROPERTIES
    MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")

# The manifest is a resource of janus_launcher.rc: no second one from the linker (they would
# clash). The client is large-address-aware: with this flag the launcher's memory scan also
# reaches client memory above 2 GB.
if(MSVC)
    target_link_options(JanusLauncher PRIVATE /MANIFEST:NO /LARGEADDRESSAWARE)
endif()
